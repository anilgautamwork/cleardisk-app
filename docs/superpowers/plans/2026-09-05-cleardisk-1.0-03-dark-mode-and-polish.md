# ClearDisk 1.0 — Plan 03: Dark Mode and UI Polish

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every screen follows the system appearance with the approved palette intact in light mode, an Appearance setting (System / Light / Dark), a button system with hover, press, disabled and busy states, and the small polish items from the spec.

**Architecture:** `UI.swift` stays the single palette; each token becomes a light/dark pair through `Color(light:dark:)` backed by a dynamic `NSColor`. View files use tokens only, enforced by a build-script guard. Button styles gain hover/press state through a nested `Body` view. Settings is a SwiftUI `Settings` scene with one picker.

**Tech Stack:** SwiftUI, AppKit (`NSColor(name:dynamicProvider:)`, `NSAppearance`), XCTest.

**Spec:** `docs/superpowers/specs/2026-09-05-cleardisk-1.0-production-design.md` (section 7)

## Global Constraints

- After Task 1, `Color(hex:` and `background(.white` may appear only in `Sources/ClearDiskApp/UI.swift`; `Scripts/make-app.sh` fails otherwise.
- Nothing calls `preferredColorScheme`. Appearance is applied through `NSApp.appearance` from one stored value `UserDefaults["appearance"]` (`system` / `light` / `dark`).
- Layout does not change; only colors, button styling and the listed polish items.
- Category and file-type accent colors stay single-valued.
- `swift build`, `swift test`, and `./Scripts/make-app.sh` green after every task; commit after every task.

```swift
final class PaletteTests: XCTestCase {
    private func rgb(_ color: NSColor, in appearance: NSAppearance.Name) -> [Int] {
        var out: [Int] = []
        NSAppearance(named: appearance)!.performAsCurrentDrawingAppearance {
            let c = color.usingColorSpace(.sRGB)!
            out = [c.redComponent, c.greenComponent, c.blueComponent].map { Int(($0 * 255).rounded()) }

        return out
    }

    func testAdaptiveColorResolvesPerAppearance() {
```

```swift
extension NSColor {
    convenience init(hex: UInt32) {
        self.init(srgbRed: CGFloat((hex >> 16) & 0xFF) / 255,
                  green: CGFloat((hex >> 8) & 0xFF) / 255,
                  blue: CGFloat(hex & 0xFF) / 255, alpha: 1)

    static func adaptive(light: UInt32, dark: UInt32) -> NSColor {
```

```swift
import AppKit
import SwiftUI

enum Appearance: String, CaseIterable, Identifiable {
    case system, light, dark

    static let key = "appearance"
    var id: String { rawValue }

    var label: String {
        switch self {
        case .system: "System"
        case .light: "Light"
        case .dark: "Dark"
        }
    }

    var nsAppearance: NSAppearance? {
        switch self {
        case .system: nil
        case .light: NSAppearance(named: .aqua)
        case .dark: NSAppearance(named: .darkAqua)
        }
    }

    static var current: Appearance {
        Appearance(rawValue: UserDefaults.standard.string(forKey: key) ?? "") ?? .system
    }

    /// nil = follow System Settings, which is the default.
    @MainActor static func apply() {
        NSApp.appearance = current.nsAppearance
    }
}

struct SettingsView: View {
    @AppStorage(Appearance.key) private var appearance = Appearance.system.rawValue

    var body: some View {
        Form {
            Picker("Appearance", selection: $appearance) {
                ForEach(Appearance.allCases) { option in
                    Text(option.label).tag(option.rawValue)
                }
            }
            .pickerStyle(.segmented)
            Text("System follows the setting in System Settings → Appearance.")
                .font(.system(size: 11.5))
                .foregroundStyle(UI.textSecondary)
        }
        .padding(20)
        .frame(width: 360)
        .onChange(of: appearance) { Appearance.apply() }
    }
}
```

```swift
/// Shared hover/press/disabled/busy behaviour for the filled styles.
private struct FilledButtonBody: View {
    let configuration: ButtonStyle.Configuration
    let fill: Color
    let compact: Bool
    let busy: Bool
    let shadow: Bool
    @Environment(\.isEnabled) private var isEnabled
    @State private var hovering = false

    var body: some View {
        configuration.label
            .opacity(busy ? 0 : 1)
            .overlay {
                if busy { ProgressView().controlSize(.small).tint(.white) }
            }
            .font(.system(size: compact ? 13 : 14, weight: .semibold))
            .foregroundStyle(.white)
            .padding(.horizontal, compact ? 16 : 22)
            .padding(.vertical, compact ? 8 : 11)
            .background(fill.opacity(configuration.isPressed ? 0.78 : 1), in: RoundedRectangle(cornerRadius: 10))
            .brightness(hovering && isEnabled && !configuration.isPressed ? 0.06 : 0)
            .shadow(color: shadow && isEnabled ? fill.opacity(0.28) : .clear, radius: 6, y: 2)
            .scaleEffect(configuration.isPressed ? 0.97 : 1)
            .opacity(isEnabled ? 1 : 0.45)
            .contentShape(RoundedRectangle(cornerRadius: 10))
            .onHover { hovering = $0 }
            .animation(.easeOut(duration: 0.12), value: hovering)
            .animation(.easeOut(duration: 0.1), value: configuration.isPressed)
    }
}

/// The design canvas's blue CTA.
struct PrimaryButtonStyle: ButtonStyle {
    var compact = false
    var busy = false

    func makeBody(configuration: Configuration) -> some View {
        FilledButtonBody(configuration: configuration, fill: UI.accent, compact: compact, busy: busy, shadow: true)
    }
}

/// Red, for the "Yes, delete" button only.

    var busy = false

    }
}
```

# ClearDisk 1.0 — Plan 01: License Worker (web/)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Cloudflare Worker at cleardisk.app that issues license keys from Stripe purchases, emails them, activates them for up to 3 Macs with a signed receipt, and lets buyers recover a lost key.

**Architecture:** One Worker (`web/`) serves `web/public/` as static assets and routes `/api/*` to handlers. KV namespace `LICENSES` stores one record per key plus three index entries. Keys are an HMAC of the Stripe session id, so the webhook and the thank-you page can both issue idempotently. Activation returns an Ed25519 signature the app verifies offline.

**Tech Stack:** TypeScript, wrangler 4, `stripe` npm SDK with the fetch HTTP client, vitest 3 + `@cloudflare/vitest-pool-workers`, Cloudflare KV, Cloudflare Email Sending (`send_email` binding), Stripe CLI for local webhook tests.

**Spec:** `docs/superpowers/specs/2026-09-05-cleardisk-1.0-production-design.md` (section 4)

## Global Constraints

- Key: `CLDK-XXXX-XXXX-XXXX-XXXX`, Crockford alphabet `0123456789ABCDEFGHJKMNPQRSTVWXYZ`, derived as `HMAC-SHA256(KEY_SECRET, "cleardisk:v1:" + session.id)`, first 10 bytes → 16 chars.
- Normalization: uppercase, `O→0`, `I→1`, `L→1`, strip non `[A-Z0-9]`, drop leading `CLDK`, require exactly 16 alphabet characters, re-group.
- Receipt message `"cleardisk:v1:" + key + ":" + machineId`, Ed25519, signature base64.
- `MAX_ACTIVATIONS` = 3. Email from `hello@cleardisk.app`, binding name `EMAIL`.
- Routes: `POST /api/checkout`, `POST /api/stripe/webhook`, `GET /api/key`, `POST /api/activate`, `POST /api/recover`, plus `GET /api/health`.
- Status codes for activate: 200 ok, 400 bad body, 404 unknown key, 403 revoked, 409 limit reached.
- Recover always answers 200 with the same message.
- All handlers take `(request: Request, env: Env)` so tests call them directly with a fake env; only KV comes from the test pool.
- Run tests with `cd web && npm test`. Commit after each task.

`web/src/keys.ts`:
```ts
// Crockford base32: no I, L, O, U. Confusables are mapped on input.
export const KEY_ALPHABET = '0123456789ABCDEFGHJKMNPQRSTVWXYZ';
const KEY_PREFIX = 'CLDK';
const CORE_LENGTH = 16;

export function crockford(bytes: Uint8Array): string {
  let out = '';
  let value = 0;
  let bits = 0;
  for (const byte of bytes) {
    value = (value << 8) | byte;
    bits += 8;
    while (bits >= 5) {
      out += KEY_ALPHABET[(value >>> (bits - 5)) & 31];
      bits -= 5;
      value &= (1 << bits) - 1;
    }
  }
  if (bits > 0) out += KEY_ALPHABET[(value << (5 - bits)) & 31];
  return out;
}

export function formatKey(core: string): string {
  return `${KEY_PREFIX}-${core.slice(0, 4)}-${core.slice(4, 8)}-${core.slice(8, 12)}-${core.slice(12, 16)}`;
}

/** HMAC of the Stripe session id → canonical key. Deterministic, so re-issuing is idempotent. */
export async function deriveKey(secret: string, sessionId: string): Promise<string> {
  const enc = new TextEncoder();
  const hmacKey = await crypto.subtle.importKey('raw', enc.encode(secret), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  const mac = new Uint8Array(await crypto.subtle.sign('HMAC', hmacKey, enc.encode(`cleardisk:v1:${sessionId}`)));
  return formatKey(crockford(mac.subarray(0, 10))); // 80 bits → exactly 16 chars
}

/** Any pasted text → canonical key, or null when it cannot be one. */
export function normalizeKey(input: string): string | null {
  let s = input.toUpperCase().replace(/O/g, '0').replace(/[IL]/g, '1').replace(/[^A-Z0-9]/g, '');
  if (s.startsWith(KEY_PREFIX)) s = s.slice(KEY_PREFIX.length);
  if (s.length !== CORE_LENGTH) return null;
  for (const ch of s) if (!KEY_ALPHABET.includes(ch)) return null;
  return formatKey(s);
}
```

`web/src/receipt.ts`:
```ts
export function toBase64(bytes: Uint8Array): string {
  let bin = '';
  for (const b of bytes) bin += String.fromCharCode(b);
  return btoa(bin);
}

export function fromBase64(text: string): Uint8Array {
  const bin = atob(text);
  const out = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
  return out;
}

export function receiptMessage(key: string, machineId: string): string {
  return `cleardisk:v1:${key}:${machineId}`;
}

export async function signReceipt(pkcs8Base64: string, key: string, machineId: string): Promise<string> {
  const privateKey = await crypto.subtle.importKey('pkcs8', fromBase64(pkcs8Base64), { name: 'Ed25519' }, false, ['sign']);
  const sig = await crypto.subtle.sign('Ed25519', privateKey, new TextEncoder().encode(receiptMessage(key, machineId)));
  return toBase64(new Uint8Array(sig));
}

export async function verifyReceipt(publicKeyRawBase64: string, key: string, machineId: string, signatureBase64: string): Promise<boolean> {
  try {
    const publicKey = await crypto.subtle.importKey('raw', fromBase64(publicKeyRawBase64), { name: 'Ed25519' }, false, ['verify']);
    return await crypto.subtle.verify('Ed25519', publicKey, fromBase64(signatureBase64), new TextEncoder().encode(receiptMessage(key, machineId)));
  } catch {
    return false;
  }
}
```

export interface Activation {
  machineId: string;
  machineName: string;
  activatedAt: string;
  lastSeenAt: string;
  appVersion: string;
}

export interface LicenseRecord {
  key: string;
  email: string;
  sessionId: string;
  paymentIntent: string;
  status: 'active' | 'revoked';
  createdAt: string;
  revokedAt?: string;
  activations: Activation[];
}

export interface PaidSession {
  id: string;
  payment_intent: string | null;
  customer_details: { email: string | null } | null;
}

export async function getLicense(env: Env, key: string): Promise<LicenseRecord | null> {
  return (await env.LICENSES.get(`key:${key}`, 'json')) as LicenseRecord | null;
}

export async function putLicense(env: Env, record: LicenseRecord): Promise<void> {
  await env.LICENSES.put(`key:${record.key}`, JSON.stringify(record));
}

/**
 * Create the license for a paid checkout session, or return the existing one.
 * ponytail: the webhook and the thanks page can race; the key is deterministic
 * so the worst case is one duplicate email, never two keys.
 */
export async function issueKey(env: Env, session: PaidSession): Promise<LicenseRecord> {
  const key = await deriveKey(env.KEY_SECRET, session.id);
  const existing = await getLicense(env, key);
  if (existing) return existing;

  const email = (session.customer_details?.email ?? '').trim().toLowerCase();
  const record: LicenseRecord = {
    key,
    email,
    sessionId: session.id,
    paymentIntent: String(session.payment_intent ?? ''),
    status: 'active',
    createdAt: new Date().toISOString(),
    activations: [],
  };
  await Promise.all([
    putLicense(env, record),
    env.LICENSES.put(`email:${email}`, key),
    env.LICENSES.put(`pi:${record.paymentIntent}`, key),
    env.LICENSES.put(`session:${session.id}`, key),
  ]);
  if (email) await sendKeyEmail(env, email, key);
  return record;
}

export async function revokeByPaymentIntent(env: Env, paymentIntent: string): Promise<boolean> {
  const key = await env.LICENSES.get(`pi:${paymentIntent}`);
  if (!key) return false;
  const record = await getLicense(env, key);
  if (!record) return false;
  if (record.status !== 'revoked') {
    record.status = 'revoked';
    record.revokedAt = new Date().toISOString();
    await putLicense(env, record);
  }
  return true;
}

import { sendKeyEmail } from './email';
import type { Env } from './env';
import { json } from './env';
import { normalizeKey } from './keys';
import { signReceipt } from './receipt';
import { getLicense, putLicense } from './store';

interface ActivateBody { key?: unknown; machineId?: unknown; machineName?: unknown; appVersion?: unknown }

export async function handleActivate(request: Request, env: Env): Promise<Response> {
  let body: ActivateBody;
  try { body = await request.json() as ActivateBody; } catch { return json({ error: 'bad_json' }, 400); }

  const key = typeof body.key === 'string' ? normalizeKey(body.key) : null;
  const machineId = typeof body.machineId === 'string' ? body.machineId.trim() : '';
  const machineName = typeof body.machineName === 'string' ? body.machineName.trim().slice(0, 80) : '';
  const appVersion = typeof body.appVersion === 'string' ? body.appVersion.slice(0, 20) : '';
  if (!key || machineId.length < 8 || !machineName) return json({ error: 'bad_request' }, 400);

  const record = await getLicense(env, key);
  if (!record) return json({ error: 'unknown_key' }, 404);
  if (record.status === 'revoked') return json({ error: 'revoked' }, 403);

  const now = new Date().toISOString();
  let activation = record.activations.find((a) => a.machineId === machineId);
  if (activation) {
    activation.lastSeenAt = now;
    activation.appVersion = appVersion;
    activation.machineName = machineName;
  } else {
    const max = Number(env.MAX_ACTIVATIONS) || 3;
    if (record.activations.length >= max) {
      return json({ error: 'limit', machines: record.activations.map((a) => a.machineName) }, 409);
    }
    activation = { machineId, machineName, activatedAt: now, lastSeenAt: now, appVersion };
    record.activations.push(activation);
  }
  await putLicense(env, record);

  const signature = await signReceipt(env.LICENSE_SIGNING_KEY, key, machineId);
  return json({ key, email: record.email, machineId, activatedAt: activation.activatedAt, signature });
}

const RECOVER_MESSAGE = "If we have a license for that address, it's on its way.";

export async function handleRecover(request: Request, env: Env): Promise<Response> {
  let email = '';
  try {
    const body = await request.json() as { email?: unknown };
    email = typeof body.email === 'string' ? body.email.trim().toLowerCase() : '';
  } catch { /* fall through: same answer either way */ }
  if (email && email.includes('@')) {
    const key = await env.LICENSES.get(`email:${email}`);
    const record = key ? await getLicense(env, key) : null;
    if (record && record.status === 'active') await sendKeyEmail(env, email, record.key);
  }
  return json({ ok: true, message: RECOVER_MESSAGE });
}

import Stripe from 'stripe';
import type { Env } from './env';
import { json } from './env';
import { getLicense, issueKey, revokeByPaymentIntent } from './store';

export function stripeClient(env: Env): Stripe {
  return new Stripe(env.STRIPE_SECRET_KEY, { httpClient: Stripe.createFetchHttpClient() });
}

const SESSION_ID = /^cs_(test|live)_[A-Za-z0-9]+$/;

/** Embedded Checkout: the page mounts Stripe's form with this client secret. */
export async function handleCheckout(request: Request, env: Env): Promise<Response> {
  let reclaim = '';
  try {
    const body = await request.json() as { reclaim?: unknown };
    if (typeof body.reclaim === 'string') reclaim = body.reclaim.slice(0, 16);
  } catch { /* body is optional */ }

  const session = await stripeClient(env).checkout.sessions.create({
    ui_mode: 'embedded',
    mode: 'payment',
    line_items: [{ price: env.STRIPE_PRICE_ID, quantity: 1 }],
    return_url: `${env.SITE_URL}/thanks?session_id={CHECKOUT_SESSION_ID}`,
    automatic_tax: { enabled: env.AUTOMATIC_TAX === 'true' },
    metadata: { reclaim },
  });
  return json({ clientSecret: session.client_secret });
}

export async function handleWebhook(request: Request, env: Env): Promise<Response> {
  const signature = request.headers.get('stripe-signature') ?? '';
  const payload = await request.text();
  let event: Stripe.Event;
  try {
    event = await stripeClient(env).webhooks.constructEventAsync(
      payload, signature, env.STRIPE_WEBHOOK_SECRET, undefined, Stripe.createSubtleCryptoProvider(),
    );
  } catch (err) {
    console.warn('webhook signature rejected', (err as Error).message);
    return json({ error: 'bad_signature' }, 400);
  }

  switch (event.type) {
    case 'checkout.session.completed': {
      const s = event.data.object as Stripe.Checkout.Session;
      if (s.payment_status === 'paid') {
        await issueKey(env, { id: s.id, payment_intent: typeof s.payment_intent === 'string' ? s.payment_intent : s.payment_intent?.id ?? null, customer_details: s.customer_details });
      }
      break;
    }
    case 'charge.refunded': {
      const c = event.data.object as Stripe.Charge;
      if (c.refunded) await revokeByPaymentIntent(env, typeof c.payment_intent === 'string' ? c.payment_intent : c.payment_intent?.id ?? '');
      break;
    }
    case 'charge.dispute.created': {
      const d = event.data.object as Stripe.Dispute;
      await revokeByPaymentIntent(env, typeof d.payment_intent === 'string' ? d.payment_intent : d.payment_intent?.id ?? '');
      break;
    }
    default:
      break;
  }
  return json({ received: true });
}

/** Thanks page: key for a checkout session; issues it if the webhook hasn't yet. */
export async function handleKey(_request: Request, env: Env, url: URL): Promise<Response> {
  const id = url.searchParams.get('session_id') ?? '';
  if (!SESSION_ID.test(id)) return json({ error: 'bad_session' }, 400);

  let key = await env.LICENSES.get(`session:${id}`);
  if (!key) {
    const s = await stripeClient(env).checkout.sessions.retrieve(id);
    if (s.payment_status !== 'paid') return json({ error: 'unpaid' }, 404);
    key = (await issueKey(env, { id: s.id, payment_intent: typeof s.payment_intent === 'string' ? s.payment_intent : s.payment_intent?.id ?? null, customer_details: s.customer_details })).key;
  }
  const record = await getLicense(env, key);
  if (!record) return json({ error: 'unknown' }, 404);
  return json({ key, email: record.email });
}

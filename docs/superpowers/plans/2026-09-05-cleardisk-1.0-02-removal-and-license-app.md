```swift
    /// Permanent deletion of an arbitrary allowed path. Together with
    /// `purgeFromTrash` these are the ONLY call sites of FileManager.removeItem
    /// in the app (the release script enforces this). UI gates both behind a
    /// "No, keep it"-default confirmation; there is no undo.
    /// Same deny-by-default blocklist as trashing.
    public static func deleteForever(_ path: String) throws {
        if case .refused(let reason) = verdict(forTrashing: path) {
            throw TrashError.refused(reason)
        }
        try FileManager.default.removeItem(atPath: path)
    }

    /// Remove items that are already in ~/.Trash for good — the "Also remove
    /// from Trash now" checkbox and Empty Trash. Refuses anything that does not
    /// resolve to a child of ~/.Trash, so a stray URL can never reach user files.
    public static func purgeFromTrash(_ urls: [URL],
                                      home: String = NSHomeDirectory()) -> (purged: Int, failed: [(url: URL, reason: String)]) {
        let trashPrefix = home + "/.Trash/"
        var purged = 0
        var failed: [(url: URL, reason: String)] = []
        for url in urls {
            let path = (url.path as NSString).standardizingPath
            guard path.hasPrefix(trashPrefix), path.count > trashPrefix.count else {
                failed.append((url, "Not inside the Trash"))
                continue
            }
            do {
                try FileManager.default.removeItem(atPath: path)
                purged += 1
            } catch {
                failed.append((url, error.localizedDescription))
            }
        }
        return (purged, failed)
    }

    /// Everything currently sitting in ~/.Trash (for the Empty Trash confirmation).
    public static func homeTrashContents(home: String = NSHomeDirectory()) -> [URL] {
        let dir = home + "/.Trash"
        let names = (try? FileManager.default.contentsOfDirectory(atPath: dir)) ?? []
        return names.filter { $0 != ".DS_Store" }.map { URL(fileURLWithPath: dir + "/" + $0) }
    }
```

```swift
import Core
import Foundation
import SwiftUI

// MARK: - model

/// One line in the confirmation list. A System Data row can span several
/// cache roots, hence `paths`.
struct RemovalItem: Identifiable, Sendable, Equatable {
    let name: String
    let paths: [String]
    let size: Int64
    let isDirectory: Bool
    /// Trash the CONTENTS of each path and keep the folder itself (cache roots
    /// that apps expect to exist).
    var contentsOnly = false

    var id: String { paths.joined(separator: "|") }

    init(name: String, path: String, size: Int64, isDirectory: Bool) {
        self.init(name: name, paths: [path], size: size, isDirectory: isDirectory)
    }

    init(name: String, paths: [String], size: Int64, isDirectory: Bool, contentsOnly: Bool = false) {
        self.name = name
        self.paths = paths
        self.size = size
        self.isDirectory = isDirectory
        self.contentsOnly = contentsOnly
    }
}

enum RemovalMode: Sendable, Equatable {
    case trash, forever
}

struct RemovalRequest: Identifiable, Sendable {
    let id = UUID()
    let items: [RemovalItem]
    let mode: RemovalMode
    var title: String?

    var totalBytes: Int64 { items.reduce(0) { $0 + $1.size } }

    var displayTitle: String {
        if let title { return title }
        let what = items.count == 1 ? "\"\(items[0].name)\"" : "\(items.count) items"
        switch mode {
        case .trash: return "Move \(what) (\(fmtBytes(totalBytes))) to the Trash?"
        case .forever: return "Delete \(what) (\(fmtBytes(totalBytes))) forever?"
        }
    }

    static func single(name: String, path: String, size: Int64, isDirectory: Bool, mode: RemovalMode) -> RemovalRequest {
        RemovalRequest(items: [RemovalItem(name: name, path: path, size: size, isDirectory: isDirectory)], mode: mode)
    }

    /// Clean Safely: every selected row with clean roots, contents only.
    static func systemData(rows: [SystemDataReport.Row]) -> RemovalRequest {
        let items = rows.filter { !$0.cleanRoots.isEmpty }.map {
            RemovalItem(name: $0.title, paths: $0.cleanRoots, size: $0.bytes ?? 0, isDirectory: true, contentsOnly: true)
        }
        return RemovalRequest(items: items, mode: .trash,
                              title: "Move \(items.count) group\(items.count == 1 ? "" : "s") to the Trash?")
    }

    static func devJunk(_ junk: [DevJunkItem]) -> RemovalRequest {
        let items = junk.map { RemovalItem(name: $0.title, path: $0.path, size: $0.bytes, isDirectory: true) }
        return RemovalRequest(items: items, mode: .trash,
                              title: "Move \(items.count) folder\(items.count == 1 ? "" : "s") to the Trash?")
    }

    /// Empty Trash: every child of the scanned .Trash node, permanently.
    static func emptyTrash(node: FileNode, home: String) -> RemovalRequest {
        let items = (node.children ?? []).map {
            RemovalItem(name: $0.name, path: home + "/.Trash/" + $0.name, size: $0.size, isDirectory: $0.isDirectory)
        }
        return RemovalRequest(items: items, mode: .forever, title: "Empty the Trash (\(fmtBytes(node.size)))?")
    }
}

struct RemovalFailure: Sendable, Equatable {
    let name: String
    let reason: String
}

struct RemovalOutcome: Sendable {
    var trashed: [TrashService.TrashedItem] = []
    /// Still sitting in the Trash — Undo can bring it back.
    var trashedBytes: Int64 = 0
    /// Gone from disk: deleted forever or purged from the Trash.
    var freedBytes: Int64 = 0
    /// Paths to detach from the in-memory tree.
    var removedPaths: [String] = []
    var failures: [RemovalFailure] = []
    var purged = false
}

// MARK: - button rule

enum RemovalButtons {
    struct Layout: Equatable {
        let confirmTitle: String
        let cancelTitle: String
        let cancelIsDefault: Bool
        let irreversible: Bool
    }

    /// Reversible → confirm is the default. Irreversible → "No, keep it" is
    /// the default and the red button needs a deliberate click.
    static func layout(mode: RemovalMode, purge: Bool) -> Layout {
        if mode == .trash && !purge {
            return Layout(confirmTitle: "Move to Trash", cancelTitle: "Cancel", cancelIsDefault: false, irreversible: false)
        }
        return Layout(confirmTitle: "Yes, delete", cancelTitle: "No, keep it", cancelIsDefault: true, irreversible: true)
    }
}

// MARK: - executor

/// Performs the file operations. Pure function of its inputs; call it off the
/// main actor.
enum RemovalExecutor {
    static func run(_ request: RemovalRequest, purge: Bool, home: String = NSHomeDirectory(),
                    progress: (Int, Int) -> Void = { _, _ in }) -> RemovalOutcome {
        var out = RemovalOutcome()
        let trashPrefix = home + "/.Trash/"

        for (index, item) in request.items.enumerated() {
            progress(index + 1, request.items.count)
            var touched = false
            for path in item.paths {
                switch request.mode {
                case .forever where path.hasPrefix(trashPrefix):
                    let r = TrashService.purgeFromTrash([URL(fileURLWithPath: path)], home: home)
                    if r.purged == 1 {
                        out.freedBytes += item.size
                        out.removedPaths.append(path)
                    } else {
                        out.failures.append(RemovalFailure(name: item.name, reason: r.failed.first?.reason ?? "Couldn't remove"))
                    }
                case .forever:
                    do {
                        try TrashService.deleteForever(path)
                        out.freedBytes += item.size
                        out.removedPaths.append(path)
                    } catch {
                        out.failures.append(RemovalFailure(name: item.name, reason: error.localizedDescription))
                    }
                case .trash where item.contentsOnly:
                    let r = TrashService.trashContents(of: path)
                    out.trashed += r.trashed
                    out.removedPaths += r.trashed.map(\.originalPath)
                    touched = touched || !r.trashed.isEmpty
                    if r.failed > 0 {
                        out.failures.append(RemovalFailure(name: item.name, reason: "\(r.failed) item\(r.failed == 1 ? "" : "s") in use or protected"))
                    }
                case .trash:
                    do {
                        if let url = try TrashService.trash(path) {
                            out.trashed.append(TrashService.TrashedItem(originalPath: path, trashURL: url))
                            out.removedPaths.append(path)
                            touched = true
                        }
                    } catch {
                        out.failures.append(RemovalFailure(name: item.name, reason: error.localizedDescription))
                    }
                }
            }
            // ponytail: sizes are known per item, not per trashed entry; a row
            // that partially trashed still reports its whole size.
            if request.mode == .trash && touched { out.trashedBytes += item.size }
        }

        if purge, !out.trashed.isEmpty {
            let r = TrashService.purgeFromTrash(out.trashed.map(\.trashURL), home: home)
            out.failures += r.failed.map { RemovalFailure(name: $0.url.lastPathComponent, reason: $0.reason) }
            out.freedBytes += out.trashedBytes
            out.trashedBytes = 0
            out.trashed = []
            out.purged = true
        }
        return out
    }
}
```

```swift
// MARK: - view

/// Owns the sheet's stage so the detached executor can report back without
/// capturing the view struct.
@MainActor
@Observable
final class RemovalModel {
    enum Stage {
        case confirm
        case working(done: Int, total: Int)
        case result(RemovalOutcome)
    }

    var stage: Stage = .confirm

    var isWorking: Bool {
        if case .working = stage { return true }
        return false
    }

    func execute(_ request: RemovalRequest, purge: Bool, state: AppState) {
        stage = .working(done: 0, total: request.items.count)
        Task.detached(priority: .userInitiated) { [self] in
            let outcome = RemovalExecutor.run(request, purge: purge) { done, total in
                Task { @MainActor in self.stage = .working(done: done, total: total) }
            }
            await MainActor.run {
                state.finishRemoval(outcome, mode: request.mode)
                self.stage = .result(outcome)
            }
        }
    }
}
```

```swift
enum LicenseKey {
    static let alphabet: Set<Character> = Set("0123456789ABCDEFGHJKMNPQRSTVWXYZ")

    /// Mirror of the Worker's normalizeKey: canonical CLDK-XXXX-XXXX-XXXX-XXXX
    /// or nil. Uppercases, maps O→0 and I/L→1, strips everything else.
    static func normalize(_ input: String) -> String? {
        var s = input.uppercased()
            .replacingOccurrences(of: "O", with: "0")
            .replacingOccurrences(of: "I", with: "1")
            .replacingOccurrences(of: "L", with: "1")
        s = String(s.filter { $0.isASCII && ($0.isLetter || $0.isNumber) })
        if s.hasPrefix("CLDK") { s.removeFirst(4) }
        guard s.count == 16, s.allSatisfy({ alphabet.contains($0) }) else { return nil }
        let chars = Array(s)
        let groups = stride(from: 0, to: 16, by: 4).map { String(chars[$0 ..< $0 + 4]) }
        return "CLDK-" + groups.joined(separator: "-")
    }
}

struct LicenseFile: Codable, Equatable {
    var key: String
    var email: String
    var machineId: String
    var activatedAt: String
    var signature: String
    var lastCheckedAt: Date?

struct ReceiptVerifier: Sendable {
    let publicKey: Curve25519.Signing.PublicKey

    init?(publicKeyBase64: String) {
        guard let data = Data(base64Encoded: publicKeyBase64),
              let key = try? Curve25519.Signing.PublicKey(rawRepresentation: data) else { return nil }
        publicKey = key
    }

    static func message(key: String, machineId: String) -> Data {
        Data("cleardisk:v1:\(key):\(machineId)".utf8)
    }

    func isValid(_ file: LicenseFile, machineId: String) -> Bool {
        guard file.machineId == machineId, let signature = Data(base64Encoded: file.signature) else { return false }
        return publicKey.isValidSignature(signature, for: Self.message(key: file.key, machineId: machineId))
    }
}

enum MachineIdentity {
    /// Hardware UUID from the IORegistry — stable across reinstalls of macOS.
    static var hardwareUUID: String {

        guard service != 0 else { return "unknown-machine" }
        defer { IOObjectRelease(service) }
        let value = IORegistryEntryCreateCFProperty(service, "IOPlatformUUID" as CFString, kCFAllocatorDefault, 0)?
            .takeRetainedValue() as? String
        return value ?? "unknown-machine"
    }

    static var name: String { Host.current().localizedName ?? "Mac" }
}

struct ActivationReceipt: Decodable, Sendable {
    let key: String
    let email: String
    let machineId: String
    let activatedAt: String
    let signature: String
}

protocol ActivationClient: Sendable {
    func activate(key: String, machineId: String, machineName: String, appVersion: String) async throws -> ActivationReceipt
}

/// Talks to the Worker. `CLEARDISK_API` in the environment points it at
/// `wrangler dev` during development.
struct LicenseClient: ActivationClient {
    var baseURL = URL(string: ProcessInfo.processInfo.environment["CLEARDISK_API"] ?? "https://cleardisk.app")!

    private struct LimitBody: Decodable { let machines: [String] }

    func activate(key: String, machineId: String, machineName: String, appVersion: String) async throws -> ActivationReceipt {
        var request = URLRequest(url: baseURL.appending(path: "api/activate"))
        request.httpMethod = "POST"
        request.timeoutInterval = 15
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try JSONEncoder().encode([
            "key": key, "machineId": machineId, "machineName": machineName, "appVersion": appVersion,
        ])
        let data: Data
        let response: URLResponse
        do {
            (data, response) = try await URLSession.shared.data(for: request)
        } catch {
            throw LicenseError.offline
        }
        switch (response as? HTTPURLResponse)?.statusCode ?? 0 {
        case 200:
            do { return try JSONDecoder().decode(ActivationReceipt.self, from: data) } catch { throw LicenseError.server(200) }
        case 404: throw LicenseError.unknownKey
        case 403: throw LicenseError.revoked
        case 409: throw LicenseError.limitReached((try? JSONDecoder().decode(LimitBody.self, from: data))?.machines ?? [])
        case let code: throw LicenseError.server(code)
        }
    }
}

// MARK: - store

@MainActor
@Observable
final class LicenseStore {
    enum State: Equatable {
        case free
        case licensed(key: String, email: String)
    }

    enum RecheckResult: Equatable {
        case skipped, stillValid, revoked
    }

    static let recheckInterval: TimeInterval = 7 * 24 * 3600

    static var defaultFileURL: URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appending(path: "ClearDisk/license.json")
    }

    private(set) var state: State = .free
    private let fileURL: URL
    private let verifier: ReceiptVerifier
    private let machineId: String
    private let client: any ActivationClient
    private var file: LicenseFile?

    var isLicensed: Bool {
        if case .licensed = state { return true }
        return false
    }

    var licensedEmail: String? {
        if case .licensed(_, let email) = state { return email }
        return nil
    }

    init(fileURL: URL = LicenseStore.defaultFileURL,
         verifier: ReceiptVerifier = .production,
         machineId: String = MachineIdentity.hardwareUUID,
         client: any ActivationClient = LicenseClient()) {
        self.fileURL = fileURL
        self.verifier = verifier
        self.machineId = machineId
        self.client = client
        load()
    }

    /// Offline check: the file's signature must verify for THIS machine.
    func load() {
        guard let data = try? Data(contentsOf: fileURL),
              let file = try? JSONDecoder().decode(LicenseFile.self, from: data),
              verifier.isValid(file, machineId: machineId) else {
            self.file = nil
            state = .free
            return
        }
        self.file = file
        state = .licensed(key: file.key, email: file.email)
    }

    func activate(rawKey: String) async throws {
        guard let key = LicenseKey.normalize(rawKey) else { throw LicenseError.invalidFormat }
        let receipt = try await client.activate(key: key, machineId: machineId,
                                                machineName: MachineIdentity.name, appVersion: Self.appVersion)
        let file = LicenseFile(key: receipt.key, email: receipt.email, machineId: receipt.machineId,
                               activatedAt: receipt.activatedAt, signature: receipt.signature, lastCheckedAt: Date())
        // A receipt that doesn't verify is a server fault, never a licensed state.
        guard verifier.isValid(file, machineId: machineId) else { throw LicenseError.server(200) }
        try save(file)
        self.file = file
        state = .licensed(key: file.key, email: file.email)
    }

    /// Weekly: re-activate this machine. Only a definitive "revoked" or
    /// "unknown key" drops the license; any network trouble changes nothing.
    func recheckIfDue(now: Date = Date()) async -> RecheckResult {
        guard var file, isLicensed else { return .skipped }
        if let last = file.lastCheckedAt, now.timeIntervalSince(last) < Self.recheckInterval { return .skipped }
        do {
            _ = try await client.activate(key: file.key, machineId: machineId,
                                          machineName: MachineIdentity.name, appVersion: Self.appVersion)
            file.lastCheckedAt = now
            try? save(file)
            self.file = file
            return .stillValid
        } catch LicenseError.revoked, LicenseError.unknownKey {
            clear()
            return .revoked
        } catch {
            return .skipped
        }
    }

    /// Overwrites the file with empty content (the repo forbids removeItem
    /// outside TrashService); an empty file fails to decode, so it's "free".
    func clear() {
        try? Data().write(to: fileURL)
        file = nil
        state = .free
    }

    private func save(_ file: LicenseFile) throws {
        try FileManager.default.createDirectory(at: fileURL.deletingLastPathComponent(), withIntermediateDirectories: true)
        try JSONEncoder().encode(file).write(to: fileURL, options: .atomic)
        try FileManager.default.setAttributes([.posixPermissions: 0o600], ofItemAtPath: fileURL.path)
    }

    private static var appVersion: String {
        Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "dev"
    }
}
```
